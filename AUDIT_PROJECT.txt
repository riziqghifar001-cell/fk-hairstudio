   STATUS UPDATE
================================================== */

async function fkUpdateStatus(id,status){

    const now=new Date().toISOString();

    const patch={
        status:status
    };

    if(status==="Dipanggil")
        patch.dipanggil_pada=now;

    if(status==="Sedang Dilayani")
        patch.mulai_pada=now;

    if(status==="Selesai")
        patch.selesai_pada=now;

    if(status==="Batal")
        patch.dibatalkan_pada=now;

    if(status==="Menunggu"){

        patch.dipanggil_pada=null;
        patch.mulai_pada=null;
        patch.selesai_pada=null;
        patch.dibatalkan_pada=null;
    }

    const {error} =
        await FKDB
            .from("bookings")
            .update(patch)
            .eq("id",id);

    if(error){

        alert(
            "Gagal memperbarui status:\n"+
            error.message
        );

        return;
    }

    await fkLoadAdmin();
}

function fkFind(id){

    return fkAdminData.find(
        x=>x.id===id
    );
}

async function fkPanggil(id){

    if(!fkFind(id)) return;

    await fkUpdateStatus(
        id,
        "Dipanggil"
    );
}

async function fkMulai(id){

    const current =
        fkFind(id);

    if(!current) return;

    const ada =
        fkAdminData.some(function(x){

            return (
                x.id!==id &&
                x.tanggal===current.tanggal &&
                x.status==="Sedang Dilayani"
            );
        });

    if(ada){

        alert(
            "Masih ada pelanggan lain yang sedang dilayani."
        );

        return;
    }

    await fkUpdateStatus(
        id,
        "Sedang Dilayani"
    );
}

async function fkSelesai(id){

    await fkUpdateStatus(
        id,
        "Selesai"
    );
}

async function fkBatal(id){

    const x=fkFind(id);

    if(!x) return;

    if(!confirm(
        "Batalkan antrean "+
        x.nomor_antrean+
        "?"
    )) return;

    await fkUpdateStatus(
        id,
        "Batal"
    );
}

async function fkBukaLagi(id){

    await fkUpdateStatus(
        id,
        "Menunggu"
    );
}

function escapeHTML(value){

    return String(value??"")
        .replace(/&/g,"&amp;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;")
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");

}


function formatTanggal(tanggal){

    if(!tanggal) return "-";

    const p=String(tanggal).split("-");

    if(p.length!==3){
        return String(tanggal);
    }

    const bulan=[
        "Januari",
        "Februari",
        "Maret",
        "April",
        "Mei",
        "Juni",
        "Juli",
        "Agustus",
        "September",
        "Oktober",
        "November",
        "Desember"
    ];

    const y=Number(p[0]);
    const m=Number(p[1]);
    const d=Number(p[2]);

    if(!y || !m || !d){
        return String(tanggal);
    }

    return d+" "+bulan[m-1]+" "+y;

}

function renderSlots(){

    const grid=
        document.getElementById("slotGrid");

    const info=
        document.getElementById("slotInfo");

    const tanggal=
        document.getElementById(
            "tanggalBooking"
        ).value;

    grid.innerHTML="";

    selectedJam="";

    if(!tanggal){

        info.textContent=
            "Pilih tanggal terlebih dahulu.";

        info.className="slot-info";

        return;

    }

    const hariIni=tanggalHariIni();

    if(tanggal<hariIni){

        info.textContent=
            "Tanggal tidak boleh sebelum hari ini.";

        info.className=
            "slot-info full";

        return;

    }

    jamOperasional.forEach(jam=>{

        const button=
            document.createElement("button");

        button.type="button";

        button.className="slot-btn";

        button.textContent=jam;

        let disabled=false;

        if(tanggal===hariIni){

            const now=new Date();

            const currentMinutes=
                now.getHours()*60+
                now.getMinutes();

            const parts=jam.split(":");

            const slotMinutes=
                Number(parts[0])*60+
                Number(parts[1]);

            if(slotMinutes<=currentMinutes){
                disabled=true;
            }

        }

        const bentrok=data.some(
            x=>
                x.jenis==="Booking" &&
                x.tanggal===tanggal &&
                x.jamBooking===jam &&
                x.status!=="Batal"
        );

        if(bentrok){
            disabled=true;
        }

        button.disabled=disabled;

        if(!disabled){

            button.onclick=()=>{

                document
                    .querySelectorAll(".slot-btn")
                    .forEach(x=>{
                        x.classList.remove(
                            "selected"
                        );
                    });

                button.classList.add(
                    "selected"
                );

                selectedJam=jam;

                info.textContent=
                    "Slot "+
                    jam+
                    " tersedia.";

                info.className=
                    "slot-info available";

                updateSteps();

            };

        }

        grid.appendChild(button);

    });

    const tersedia=
        [...grid.querySelectorAll(
            ".slot-btn:not(:disabled)"
        )].length;

    if(tersedia===0){

        info.textContent=
            "Tidak ada slot tersedia pada tanggal ini.";

        info.className=
            "slot-info full";

    }else{

        info.textContent=
            tersedia+
            " slot masih tersedia.";

        info.className=
            "slot-info available";

    }

}

function nomorBerikutnya(tanggal){

    const hari=data.filter(
        x=>x.tanggal===tanggal
    );

    let terbesar=0;

    hari.forEach(x=>{

        const m=String(
            x.nomorAntrean||""
        ).match(/^A-(\d+)$/);

        if(m){

            terbesar=Math.max(
                terbesar,
                Number(m[1])
            );

        }

    });

    return "A-"+
        String(terbesar+1)
            .padStart(3,"0");

}

function kirimWhatsApp(){

        window.currentBookingResult;

    if(!x) return;


    const pesan=
        "Halo FK Hairstudio, saya sudah mengambil antrean.%0A"+
        "Nama: "+
        encodeURIComponent(x.nama)+
        "%0A"+
        "No. Antrean: "+
        encodeURIComponent(x.nomorAntrean)+
        "%0A"+
        "Layanan: "+
        encodeURIComponent(x.layanan)+
        "%0A"+
        "Tanggal: "+
        encodeURIComponent(
            formatTanggal(x.tanggal)
        )+
        (
            x.jamBooking
            ?
            "%0AJam: "+
            encodeURIComponent(
                x.jamBooking
            )
            :
            ""
        );


    if(!ADMIN_WA){

        if(
            navigator.clipboard &&
            window.isSecureContext
        ){

            navigator.clipboard
                .writeText(
                    decodeURIComponent(
                        pesan
                    )
                )
                .catch(()=>{});

        }

        alert(
            "Nomor WhatsApp admin belum diatur.\n\n"+
            "Pesan konfirmasi sudah disiapkan. "+
            "Setelah nomor admin dimasukkan ke kode, "+
            "tombol ini akan langsung membuka WhatsApp."
        );

        return;

    }


    window.open(
        "https://wa.me/"+
        ADMIN_WA+
        "?text="+
        pesan,
        "_blank"
    );

}

async function ambilAntrean(){

    const hpRaw =
        document.getElementById("hp")?.value.trim();

    const nama =
        document.getElementById("nama")?.value.trim();

    const service =
        document.getElementById("layananSelect")?.value.split("|");

    const jenis = getJenis();

    if(!hpRaw){

        alert(
            "Masukkan nomor WhatsApp terlebih dahulu."
        );

        return;
    }

    const hp = fkNormHP(hpRaw);

    if(hp.length < 8){

        alert(
            "Nomor WhatsApp tidak valid."
        );

        return;
    }

    if(!nama){

        alert(
            "Nama pelanggan belum diisi."
        );

        return;
    }

    if(!service || !service[0]){

        alert(
            "Silakan pilih layanan."
        );

        return;
    }

    let tanggal = fkToday();
    let jamBooking = null;

    if(jenis === "Booking"){

        tanggal =
            document.getElementById(
                "tanggalBooking"
            )?.value;

        jamBooking = fkSelectedJam;

        if(!tanggal){

            alert(
                "Silakan pilih tanggal booking."
            );

            return;
        }

        if(!jamBooking){

            alert(
                "Silakan pilih salah satu jam yang tersedia."
            );

            return;
        }
    }

    const harga = Number(service[1]) || 0;

    const submit =
        document.getElementById("submitText");

    if(submit){

        submit.textContent =
            "MENYIMPAN...";
    }

    try{

        const {data,error} =
            await FKDB.rpc(
                "create_booking",
                {
                    p_nama: nama,
                    p_hp: hp,
                    p_layanan: service[0],
                    p_harga: harga,
                    p_jenis: jenis,
                    p_tanggal: tanggal,
                    p_jam_booking: jamBooking

                message.toLowerCase().includes("slot") ||

            Array.isArray(data)
                ? data[0]
                : data;

            result?.nomor_antrean ||
            result?.nomorAntrean;

            nomorAntrean: nomor,
            nama: nama,
            hp: hp,
            layanan: service[0],
            harga: harga,

            document.getElementById("hasil");

        if(hasil){

            hasil.innerHTML =
                '<div class="result-success">' +
                '? BERHASIL DISIMPAN' +
                '</div>' +

                    jenis === "Booking"

                ) +
                '</div>' +

                '<div class="queue-number">' +
                fkEscape(nomor) +
                '</div>' +

                '<div class="result-detail">' +
                "Sampai jumpa, " +
                fkEscape(nama) +
                "</div>" +

                '<div class="result-ticket">' +

                '<div class="ticket-item">' +
                '<small>LAYANAN</small>' +
                '<strong>' +
                fkEscape(service[0]) +
                '</strong>' +
                '</div>' +

                '<div class="ticket-item">' +
                '<small>HARGA</small>' +
                '<strong>' +
                fkRupiah(harga) +
                '</strong>' +
                '</div>' +

                '<div class="ticket-item">' +
                '<small>TANGGAL</small>' +
                '<strong>' +
                fkEscape(
                    fkFormatTanggal(tanggal)
                ) +
                '</strong>' +
                '</div>' +

                '<div class="ticket-item">' +
                '<small>' +
                (
                    jenis === "Booking"
                    ? "JAM BOOKING"
                    : "JENIS"
                ) +
                '</small>' +

                '<strong>' +
                fkEscape(
                    jenis === "Booking"
                    ? jamBooking
                    : "Walk-in"
                ) +
                '</strong>' +

                '</div>' +

                '</div>' +

                '<div class="result-actions">' +

                '<button class="btn-wa" onclick="kirimWhatsApp()">' +
                'KONFIRMASI WHATSAPP' +
                '</button>' +

                '<button class="btn-reset" onclick="resetForm()">' +
                'SELESAI' +
                '</button>' +

                '</div>';

            hasil.scrollIntoView({
                behavior:"smooth",
                block:"center"
            });
        }

        document.getElementById("hp").value = "";
        document.getElementById("nama").value = "";

        const customerInfo =
            document.getElementById("customerInfo");

        if(customerInfo){
            customerInfo.style.display = "none";
        }

        if(jenis === "Booking"){

                document.getElementById("tanggalBooking");

            if(tanggalInput){

            "Terjadi kesalahan saat menghubungkan ke server."
        );

    }finally{

        if(submit){

            submit.textContent =
                "AMBIL ANTREAN";
        }
    }
}

Number(p[0]),
Number(p[1])-1,
Number(p[2])
);

}

if(!tanggal) return "-";

const bulan=[

];

function fkFormatTanggal(t){

    if(!t) return "-";

    const p=String(t).split("-");

    if(p.length!==3) return t;

    const bulan=[
        "Januari","Februari","Maret","April","Mei","Juni",
        "Juli","Agustus","September","Oktober","November","Desember"
    ];

    return Number(p[2])+" "+
        bulan[Number(p[1])-1]+" "+p[0];

function fkRange(){

    const mode =
        document.getElementById("periode")
            ?.value || "hari";

    const today=fkToday();

    if(mode==="hari")
        return [today,today];

    const now=new Date();

    if(mode==="kemarin"){

        now.setDate(now.getDate()-1);

        const x=
            now.getFullYear()+"-"+
            String(now.getMonth()+1).padStart(2,"0")+"-"+
            String(now.getDate()).padStart(2,"0");

        return [x,x];
    }

    if(mode==="minggu"){

        now.setDate(now.getDate()-6);

        const x=
            now.getFullYear()+"-"+
            String(now.getMonth()+1).padStart(2,"0")+"-"+
            String(now.getDate()).padStart(2,"0");

        return [x,today];
    }

    if(mode==="bulan"){

        const x=
            now.getFullYear()+"-"+
            String(now.getMonth()+1).padStart(2,"0")+
            "-01";

        return [x,today];
    }

    if(mode==="custom"){

        const x=
            document.getElementById("tanggal")
                ?.value || today;

        return [x,x];
    }

    return ["0000-01-01","9999-12-31"];
}

function fkEscape(v){

function fkFormatTanggal(t){

async function fkLoadCustomers(){

    const {data,error} =
        await FKDB
            .from("bookings")
            .select("*")
            .order("dibuat_pada",{ascending:false});

    if(error){

        alert(
            "Gagal membaca pelanggan:\n"+
            error.message
        );

        return;
    }

    fkCustomers=data||[];

    fkRenderCustomers();
}

function fkRenderCustomers(){

    const grouped={};

    fkCustomers.forEach(function(x){

        const phone=fkNorm(x.hp);

        if(!phone) return;

        if(!grouped[phone])
            grouped[phone]=[];

        grouped[phone].push(x);
    });

    const search =
        document.getElementById("search")
            ?.value
            .toLowerCase()
            .trim() || "";

    const list =
        Object.entries(grouped)
            .filter(function(entry){

                const phone=entry[0];
                const records=entry[1];

                const name =
                    String(
                        records[0]?.nama || ""
                    ).toLowerCase();

                return (
                    !search ||
                    phone.includes(
                        search.replace(/\D/g,"")
                    ) ||
                    name.includes(search)
                );
            })
            .sort(function(a,b){

                return String(
                    b[1][0]?.dibuat_pada || ""
                ).localeCompare(
                    String(
                        a[1][0]?.dibuat_pada || ""
                    )
                );
            });

    const grid =
        document.getElementById(
            "customerGrid"
        );

    if(!grid) return;

    if(!list.length){

        grid.innerHTML =
            '<div class="empty">' +
            'Belum ada data pelanggan.' +
            '</div>';

        return;
    }

    grid.innerHTML =
        list.map(function(entry){

            const phone=entry[0];
            const records=entry[1];

            const selesai =
                records.filter(
                    x=>x.status==="Selesai"
                );

            const spend =
                selesai.reduce(
                    (a,b)=>
                        a+Number(b.harga||0),
                    0
                );

            return (
                '<div class="customer-card" ' +
                'onclick="fkOpenCustomer(\''+
                phone+
                '\')">' +

                '<div class="customer-name">' +
                fkEscape(
                    records[0]?.nama || "-"
                ) +
                '</div>' +

                '<div class="customer-phone">' +
                fkEscape(phone) +
                '</div>' +

                '<div class="customer-meta">' +
                records.length+
                ' kunjungan • '+
                fkRupiah(spend) +
                '</div>' +

                '</div>'
            );

        }).join("");
}
